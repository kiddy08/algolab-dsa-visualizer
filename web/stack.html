1. Java Code

Stack.java 
class Stack {
    private int[] data;
    private int top = -1;

    Stack(int capacity) {
        data = new int[capacity];
    }

    void push(int value) {
        if (top == data.length - 1)
            throw new RuntimeException("Stack Overflow");
        data[++top] = value;
    }

    int pop() {
        if (top == -1)
            throw new RuntimeException("Stack Underflow");
        return data[top--];
    }

    int peek() {
        if (top == -1)
            throw new RuntimeException("Stack is empty");
        return data[top];
    }

    boolean isEmpty() {
        return top == -1;
    }

    int size() {
        return top + 1;
    }
}

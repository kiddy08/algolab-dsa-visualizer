1. Java Code

Deque.java 
class Deque {
    private int[] data;
    private int front = 0;
    private int size = 0;

    Deque(int capacity) {
        data = new int[capacity];
    }

    void addFront(int value) {
        if (size == data.length)
            throw new RuntimeException("Deque Overflow");
        front = (front - 1 + data.length) % data.length;
        data[front] = value;
        size++;
    }

    void addRear(int value) {
        if (size == data.length)
            throw new RuntimeException("Deque Overflow");
        int rear = (front + size) % data.length;
        data[rear] = value;
        size++;
    }

    int removeFront() {
        if (size == 0)
            throw new RuntimeException("Deque Underflow");
        int value = data[front];
        front = (front + 1) % data.length;
        size--;
        return value;
    }

    int removeRear() {
        if (size == 0)
            throw new RuntimeException("Deque Underflow");
        int rear = (front + size - 1) % data.length;
        int value = data[rear];
        size--;
        return value;
    }

    int peekFront() {
        if (size == 0)
            throw new RuntimeException("Deque is empty");
        return data[front];
    }

    int peekRear() {
        if (size == 0)
            throw new RuntimeException("Deque is empty");
        return data[(front + size - 1) % data.length];
    }

    boolean isEmpty() {
        return size == 0;
    }

    int size() {
        return size;
    }
}

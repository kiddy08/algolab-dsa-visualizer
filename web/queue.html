1. Java Code

Queue.java 
class Queue {
    private int[] data;
    private int front = 0;
    private int rear = -1;

    Queue(int capacity) {
        data = new int[capacity];
    }

    void enqueue(int value) {
        if (rear == data.length - 1)
            throw new RuntimeException("Queue Overflow");
        data[++rear] = value;
    }

    int dequeue() {
        if (front > rear)
            throw new RuntimeException("Queue Underflow");
        int value = data[front++];
        if (front > rear) {
            front = 0;
            rear = -1;
        }
        return value;
    }

    int peek() {
        if (front > rear)
            throw new RuntimeException("Queue is empty");
        return data[front];
    }

    boolean isEmpty() {
        return front > rear;
    }

    int size() {
        return rear - front + 1;
    }
}

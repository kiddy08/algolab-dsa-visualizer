1. Java Code

Trie.java 
class TrieNode {
    Map<Character, TrieNode> children = new TreeMap<>();
    boolean isEndOfWord = false;
}

void insert(String word) {
    TrieNode curr = root;
    for (char ch : word.toCharArray()) {
        curr.children.putIfAbsent(ch, new TrieNode());
        curr = curr.children.get(ch);
    }
    curr.isEndOfWord = true;
}

boolean search(String word) {
    TrieNode node = walkPrefix(word);
    return node != null && node.isEndOfWord;
}

boolean startsWith(String prefix) {
    return walkPrefix(prefix) != null;
}

TrieNode walkPrefix(String str) {
    TrieNode curr = root;
    for (char ch : str.toCharArray()) {
        if (!curr.children.containsKey(ch)) return null;
        curr = curr.children.get(ch);
    }
    return curr;
}

boolean delete(TrieNode curr, String word, int idx) {
    if (idx == word.length()) {
        if (!curr.isEndOfWord) return false;
        curr.isEndOfWord = false;
        return curr.children.isEmpty();
    }
    char ch = word.charAt(idx);
    TrieNode next = curr.children.get(ch);
    if (next == null) return false;
    if (delete(next, word, idx + 1)) {
        curr.children.remove(ch);
        return !curr.isEndOfWord && curr.children.isEmpty();
    }
    return false;
}

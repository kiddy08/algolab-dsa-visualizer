1. Java Code

MinHeap.java 
class BinaryHeap {
    int[] heap = new int[15];
    int size = 0;

    void insert(int val) {
        if (size == heap.length) throw new RuntimeException("Full");
        int i = size++;
        heap[i] = val;
        while (i > 0 && outOfOrder(parent(i), i)) {
            swap(i, parent(i));
            i = parent(i);
        }
    }

    int extractRoot() {
        if (size == 0) throw new RuntimeException("Empty");
        int root = heap[0];
        heap[0] = heap[--size];
        siftDown(0);
        return root;
    }

    void siftDown(int i) {
        while (2 * i + 1 < size) {
            int best = 2 * i + 1, right = 2 * i + 2;
            if (right < size && outOfOrder(best, right))
                best = right;
            if (!outOfOrder(i, best)) break;
            swap(i, best);
            i = best;
        }
    }

    int peek() {
        if (size == 0) throw new RuntimeException("Empty");
        return heap[0];
    }

    void buildHeap() {
        for (int i = size / 2 - 1; i >= 0; i--)
            siftDown(i);
    }
}
